#======== Controller
from odoo import http
from odoo.http import request


class SmsController(http.Controller):

    @http.route(
        '/shop/send_order_sms',
        type='json',
        auth='public',
        website=True,
        csrf=False
    )
    def send_order_sms(self):
        """
        Envoi SMS client immédiat
        + planification SMS admin (+1 min)
        Sécurisé par la session website
        """
        order = request.website.sale_get_order()
        if not order:
            return {'error': 'Aucune commande en session'}

        return order.send_order_sms()


#============== Model
from odoo import models, fields, _
from datetime import timedelta
import logging

_logger = logging.getLogger(__name__)


class SaleOrder(models.Model):
    _inherit = 'sale.order'

    sms_client_sent = fields.Boolean(
        string="SMS client envoyé",
        default=False
    )
    sms_admin_sent = fields.Boolean(
        string="SMS admin envoyé",
        default=False
    )
    sms_admin_scheduled_at = fields.Datetime(
        string="SMS admin planifié le"
    )

    # ======================================================
    # MÉTHODE APPELÉE PAR LE CONTROLLER
    # ======================================================
    def send_order_sms(self):
        self.ensure_one()

        if self.sms_client_sent:
            return {'error': 'SMS déjà envoyé pour cette commande'}

        if self.state not in ('sale', 'done'):
            return {'error': 'Commande non confirmée'}

        config = self.env['dexchange.sms.config'].sudo().search(
            [('active', '=', True)], limit=1
        )
        if not config:
            return {'error': 'Configuration SMS introuvable'}

        sms_service = self.env['dexchange.sms.service'].sudo()

        # =============================
        # 📩 SMS CLIENT (IMMÉDIAT)
        # =============================
        phone = self.partner_id.mobile or self.partner_id.phone
        if not phone:
            return {'error': 'Numéro client manquant'}

        message_client = _(
            "Votre commande %(order)s est confirmée. Merci de votre achat !"
        ) % {'order': self.name}

        try:
            res = sms_service.send_sms(
                recipients=[phone],
                content=message_client,
                sender=config.sender,
                api_key=config.api_key,
                url=config.url
            )
        except Exception as e:
            _logger.exception("Erreur SMS client")
            return {'error': str(e)}

        if res.get('error'):
            return {'error': res.get('error')}

        self.sms_client_sent = True

        # =============================
        # ⏱ PLANIFIER SMS ADMIN (+1 min)
        # =============================
        self.sms_admin_scheduled_at = fields.Datetime.now() + timedelta(minutes=1)

        return {
            'success': True,
            'message': 'SMS client envoyé, SMS admin programmé'
        }

    # ======================================================
    # MÉTHODE APPELÉE PAR LE CRON
    # ======================================================
    def send_admin_sms(self):
        for order in self:
            if order.sms_admin_sent:
                continue

            config = self.env['dexchange.sms.config'].sudo().search(
                [('active', '=', True)], limit=1
            )
            if not config:
                continue

            sms_service = self.env['dexchange.sms.service'].sudo()

            admin_phone = '221778988372'

            message_admin = _(
                "La commande %(order)s de %(customer)s est confirmée. "
                "Veuillez la traiter dans les meilleurs délais."
            ) % {
                'order': order.name,
                'customer': order.partner_id.name
            }

            try:
                res = sms_service.send_sms(
                    recipients=[admin_phone],
                    content=message_admin,
                    sender=config.sender,
                    api_key=config.api_key,
                    url=config.url
                )
            except Exception:
                _logger.exception("Erreur SMS admin")
                continue

            if not res.get('error'):
                order.sms_admin_sent = True
